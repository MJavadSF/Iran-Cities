## پیدا کردن

```ts
import { getProvince, getCities, getCity, getNeighborProvinces } from '@code-plate/iran-cities';

getProvince('تهران');           // با نام فارسی، شناسه‌ی انگلیسی یا id
getCities('fars');              // شهرهای یک استان
getCity('شيراز');               // «ي» عربی هم درست کار می‌کند
getCity('شیراز', 'fars');       // داخل یک استان
getNeighborProvinces('قم');     // [اصفهان، مرکزی، سمنان، تهران]
```

## جستجو

```ts
import { searchCities } from '@code-plate/iran-cities';

searchCities('شهر', { limit: 10 });
searchCities('تهرن', { fuzzy: true });   // تحمل غلط تایپی
searchCities('jivhk');                   // با کیبورد انگلیسی تایپ شده ← تهران
```
